cmake_minimum_required(VERSION 3.16)
project(LabLine VERSION 1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "LabLine requires Linux; use WSL2 Ubuntu on Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LABLINE_ASAN "Address + undefined behavior sanitizers" OFF)
option(LABLINE_TSAN "Thread sanitizer (separate build)" OFF)
option(LABLINE_UNSAFE_RACE "Build isolated deliberately racy example" OFF)
if(LABLINE_ASAN AND LABLINE_TSAN)
  message(FATAL_ERROR "Do not combine ASan and TSan")
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
if(LABLINE_ASAN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
elseif(LABLINE_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=thread)
endif()
find_package(Threads REQUIRED)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
add_library(labline_core src/core.cpp src/protocol.cpp src/transport.cpp src/simulator.cpp src/controller.cpp)
target_include_directories(labline_core PUBLIC include)
target_link_libraries(labline_core PUBLIC Threads::Threads)
add_executable(labline src/main.cpp)
target_link_libraries(labline PRIVATE labline_core)
add_executable(unit_tests tests/unit.cpp)
target_link_libraries(unit_tests PRIVATE labline_core)
add_executable(deadlock_demo tests/deadlock_demo.cpp)
target_link_libraries(deadlock_demo PRIVATE Threads::Threads)
add_executable(simulator_probe tests/simulator_probe.cpp)
target_link_libraries(simulator_probe PRIVATE labline_core)
if(LABLINE_UNSAFE_RACE)
  add_executable(unsafe_race_demo tests/unsafe_race_demo.cpp)
  target_link_libraries(unsafe_race_demo PRIVATE Threads::Threads)
endif()
enable_testing()
foreach(case roundtrip split_multiple crc length version state queue)
  add_test(NAME unit_${case} COMMAND unit_tests ${case})
endforeach()
foreach(case concurrency ack crc slow crash timeout repeat audit deadlock)
  add_test(NAME integration_${case} COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:labline> ${case})
endforeach()
add_test(NAME simulator_input_safety COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/device_protocol.py $<TARGET_FILE:simulator_probe>)
get_property(all_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${all_tests} PROPERTIES TIMEOUT 30)
